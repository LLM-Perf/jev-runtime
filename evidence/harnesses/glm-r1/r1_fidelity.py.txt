"""Audit saved input IDs against checkpoint bytes and create an explicit generic profile."""
import hashlib,json,shutil,time
from pathlib import Path
from tokenizers import Tokenizer
from transformers import AutoTokenizer,PreTrainedTokenizerFast
from jev_runtime.compiler import Compiler
from jev_runtime.schema import Bundle,ModelIdentity,Question

root=Path('/root/jev-runtime');model=root/'models/DeepSeek-R1-Distill-Qwen-7B'
out=root/'r1-qwen7b-tokenizer-fidelity-d1ef50e.json';assert not out.exists()
source=json.loads((model/'jev-source.json').read_text())
report={'model_id':source['model_id'],'revision':source['revision'],'started_at':time.time(),
        'harness_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        'qualification':'Exact tokenizer artifact and saved-input audit; no GPU scoring in this script'}
raw=Tokenizer.from_file(str(model/'tokenizer.json'))
generic=PreTrainedTokenizerFast.from_pretrained(str(model),trust_remote_code=False)
compiler=Compiler(generic)
bundle=Bundle(id='fidelity',version=1,model=ModelIdentity(id=source['model_id'],revision=source['revision'],
    tokenizer_digest=compiler.tokenizer_digest,tokenizer_implementation_digest=compiler.tokenizer_implementation_digest,
    template_digest=compiler.template_digest))
question=Question.model_validate({'id':'intent','type':'choice','instruction':"Select the customer's primary intent.",
    'options':[{'id':'billing','description':'Payments, charges, or refunds'},
               {'id':'technical','description':'A technical fault in the product'},
               {'id':'other','description':'A different topic'}]})
prompts=[];original=compiler._encode

def capture(prompt,labels,budget):
    prompts.append(prompt);return original(prompt,labels,budget)
compiler._encode=capture
seq=compiler.compile('I was charged twice and would like a refund.',question,bundle,'audit').sequences[0]
assert list(seq.input_ids)==raw.encode(prompts[-1],add_special_tokens=False).ids
report['expected_input_ids']=list(seq.input_ids);report['expected_label_ids']=list(seq.label_ids)
report['rendered_prompt_sha256']=hashlib.sha256(prompts[-1].encode()).hexdigest()
report['saved_serving_inputs']={}
for engine in ('vllm','sglang'):
 contract=json.loads((root/'runs'/f'{engine}-r1-qwen7b-tp4-d1ef50e/contract.json').read_text())
 actual=contract['reference_logits']
 report['saved_serving_inputs'][engine]={'input_ids':actual['input_ids'],'label_ids':actual['label_ids'],
    'matches_checkpoint_tokenizer':actual['input_ids']==list(seq.input_ids) and actual['label_ids']==list(seq.label_ids),
    'tokenizer_implementation_digest':contract['model']['tokenizer_implementation_digest']}

assert all(c['matches_checkpoint_tokenizer'] for c in report['saved_serving_inputs'].values())
report['finished_at']=time.time();out.write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n')
print(json.dumps({e:c['matches_checkpoint_tokenizer'] for e,c in report['saved_serving_inputs'].items()}))
