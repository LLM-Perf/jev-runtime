"""Rerun both native engines with the audited checkpoint ByteLevel tokenizer."""
import hashlib,json,subprocess,time
from pathlib import Path
ROOT=Path('/root/jev-runtime');RUNTIME='3377c95ca9955ae96ce7c7deb2c4d97b41b1d9c1'
HARNESS='b28bb4611b6a8142ffc37b1b12af84514a713b5f'
prior=json.loads((ROOT/'precision-followup-public-pair-3377c95.json').read_text())
assert prior['completed'] and all(a['validation']['cleanup_passed'] for a in prior['attempts'])
audit=json.loads((ROOT/'r1-llama-tokenizer-fidelity-3377c95.json').read_text());profile=Path(audit['explicit_profile']['path'])
for n,h in audit['explicit_profile']['files'].items():assert hashlib.sha256((profile/n).read_bytes()).hexdigest()==h,n
out=ROOT/'tokenizer-followup-public-pair-b28bb46.json';assert not out.exists()
report={'runtime_source_commit':RUNTIME,'harness_source_commit':HARNESS,
        'harness_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        'started_at':time.time(),'qualification':'Explicit checkpoint ByteLevel tokenizer profile','attempts':[]}
try:
 for engine in ('vllm','sglang'):
  run=ROOT/'runs'/f'{engine}-r1-llama8b-bytelevel-tp4-b28bb46';assert not run.exists()
  cmd=[str(ROOT/'envs'/engine/'bin/python'),str(ROOT/'releases'/HARNESS/'tests/integration/run_native_validation.py'),
       '--run-dir',str(run),'--engine',engine,'--model-path',str(ROOT/'models/DeepSeek-R1-Distill-Llama-8B'),
       '--tokenizer-path',str(profile),'--gpus','3,4,5,6','--port','18795' if engine=='vllm' else '18794',
       '--memory-fraction','.075' if engine=='vllm' else '.65','--reserve-mib','3072',
       '--switches','1000','--startup-timeout','300','--check-timeout','300','--reference-timeout','300',
       '--reference-atol','.15','--source-commit',HARNESS,'--runtime-source-commit',RUNTIME]
  a={'engine':engine,'run':str(run),'command':cmd};report['attempts'].append(a);out.write_text(json.dumps(report,indent=2)+'\n')
  with (ROOT/(run.name+'-controller.log')).open('x') as f:proc=subprocess.run(cmd,stdout=f,stderr=subprocess.STDOUT)
  v=json.loads((run/'validation.json').read_text());a['returncode']=proc.returncode;a['validation']=v
  c=json.loads((run/'contract.json').read_text());a['saved_input_matches_checkpoint_tokenizer']=(
      c['reference_logits']['input_ids']==audit['expected_input_ids'] and c['reference_logits']['label_ids']==audit['expected_label_ids'])
  out.write_text(json.dumps(report,indent=2)+'\n')
  print(json.dumps({'run':run.name,'input_fidelity':a['saved_input_matches_checkpoint_tokenizer'],'validation':v}),flush=True)
  assert v['cleanup_passed'] and proc.returncode==(0 if v['passed'] else 1)
 report['completed']=True
except BaseException as exc:
 report['completed']=False;report['failure']={'type':type(exc).__name__,'message':str(exc)};raise
finally:
 report['finished_at']=time.time();out.write_text(json.dumps(report,indent=2)+'\n')
