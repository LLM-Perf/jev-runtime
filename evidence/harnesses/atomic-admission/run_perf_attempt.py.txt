import hashlib, importlib, json, subprocess, sys, time
from pathlib import Path

harness, core, engine, label = sys.argv[1:]
root = Path('/root/jev-runtime')
source = root / 'releases' / harness
sys.path.insert(0, str(source))
from tests.integration.run_native_validation import save, wait_ready, topology, postcheck, cleanup

run = root / 'runs' / f'{engine}-atomic-{label}'
run.mkdir(exist_ok=False)
report = {'harness_source_commit': harness, 'runtime_source_commit': core, 'engine': engine,
          'qualification': 'colocated development measurement, not release certification',
          'wrapper_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
          'started_at': time.time(), 'stages': {}}
record = None
try:
    paths = {name: importlib.import_module(name).__file__ for name in ('jev_runtime', 'jev_'+engine)}
    assert all('/releases/'+core+'/' in p for p in paths.values()), paths
    report['import_paths'] = paths
    cmd = [sys.executable, str(source/'deployment/dsw_service.py'), 'launch', '--run-dir', str(run),
           '--engine', engine, '--model-path', str(root/'models/SmolLM2-1.7B-Instruct'),
           '--gpus', '7', '--port', '18795' if engine=='vllm' else '18794',
           '--memory-fraction', '.07' if engine=='vllm' else '.65', '--api-workers', '2']
    with (run/'launch.log').open('x') as f:
        subprocess.run(cmd, stdout=f, stderr=subprocess.STDOUT, check=True, timeout=45)
    record = json.loads((run/'process.json').read_text())
    keys = json.loads((run/'keys.json').read_text())
    report['stages']['ready'] = wait_ready(record, keys['api'], 300)
    save(run/'topology.json', topology(run, record))
    for concurrency in (1,16):
        cmd = [sys.executable, str(source/'benchmarks/run_case.py'), '--run-dir',str(run),
               '--output',str(run/f'c{concurrency}'),'--source-commit',harness,
               '--runtime-source-commit',core,'--context-tokens','256','--candidates','8',
               '--concurrency',str(concurrency),'--cache','hot','--runtime-timing',
               '--input-variants','1','--duration','10','--repeats','3','--warmup','32']
        with (run/f'c{concurrency}.log').open('x') as f:
            subprocess.run(cmd,stdout=f,stderr=subprocess.STDOUT,check=True,timeout=300)
        report['stages'][f'c{concurrency}']={'completed':True}
    report['stages']['postcheck'] = postcheck(run,record,keys['admin'],core)
    report['passed'] = True
except BaseException as exc:
    report['passed'] = False
    report['failure'] = {'type':type(exc).__name__,'message':str(exc)}
finally:
    if record is None and (run/'process.json').exists():
        record=json.loads((run/'process.json').read_text())
    if record:
        report['stages']['cleanup'] = cleanup(run,record)
        report['passed'] = report.get('passed',False) and report['stages']['cleanup']['passed']
    report['finished_at']=time.time()
    save(run/'performance-attempt.json',report)
    print(json.dumps(report),flush=True)
if not report['passed']:
    sys.exit(1)
